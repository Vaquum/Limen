7def calibrate_confidence_threshold(models: list, x_val: Any, y_val: Any, target_confidence: float = 0.8) -> tuple: 

8 

9 ''' 

10 Compute confidence threshold using validation data. 

11 

12 This function determines the optimal confidence threshold by analyzing model 

13 prediction variance on validation data. It establishes what level of prediction 

14 uncertainty corresponds to reliable vs unreliable predictions. 

15 

16 Args: 

17 models (list): List of trained models 

18 x_val (array-like): Validation features for threshold calibration 

19 y_val (array-like): Validation targets for performance evaluation 

20 target_confidence (float): Target percentage of predictions to classify as "confident" (0.0 to 1.0) 

21 

22 Returns: 

23 tuple: Confidence threshold and calibration statistics 

24 ''' 

25 

26 # Get model predictions on validation data 

27 val_preds = [] 

28 for model in models: 

29 preds = model.predict(x_val) 

30 val_preds.append(preds) 

31 

32 val_preds = np.array(val_preds) 

33 val_pred_mean = np.mean(val_preds, axis=0) 

34 val_pred_std = np.std(val_preds, axis=0) 

35 

36 # Set threshold based on validation data distribution 

37 confidence_threshold = np.percentile(val_pred_std, target_confidence * 100) 

38 

39 # Analyze calibration quality 

40 confident_mask = val_pred_std <= confidence_threshold 

41 uncertain_mask = ~confident_mask 

42 

43 # Calculate performance on each group 

44 if np.sum(confident_mask) > 0: 

45 confident_mae = mean_absolute_error(y_val[confident_mask], val_pred_mean[confident_mask]) 

46 confident_r2 = r2_score(y_val[confident_mask], val_pred_mean[confident_mask]) 

47 else: 

48 confident_mae, confident_r2 = np.nan, np.nan 

49 

50 if np.sum(uncertain_mask) > 0: 

51 uncertain_mae = mean_absolute_error(y_val[uncertain_mask], val_pred_mean[uncertain_mask]) 

52 uncertain_r2 = r2_score(y_val[uncertain_mask], val_pred_mean[uncertain_mask]) 

53 else: 

54 uncertain_mae, uncertain_r2 = np.nan, np.nan 

55 

56 overall_mae = mean_absolute_error(y_val, val_pred_mean) 

57 overall_r2 = r2_score(y_val, val_pred_mean) 

58 

59 calibration_stats = { 

60 'threshold': confidence_threshold, 

61 'confident_pct': np.mean(confident_mask) * 100, 

62 'val_overall_mae': overall_mae, 

63 'val_overall_r2': overall_r2, 

64 'val_confident_mae': confident_mae, 

65 'val_confident_r2': confident_r2, 

66 'val_uncertain_mae': uncertain_mae, 

67 'val_uncertain_r2': uncertain_r2, 

68 'val_std_stats': { 

69 'min': np.min(val_pred_std), 

70 'median': np.median(val_pred_std), 

71 'max': np.max(val_pred_std), 

72 'threshold': confidence_threshold 

73 } 

74 } 

75 

76 return confidence_threshold, calibration_stats

143def confidence_filtering_system(models: list, data: dict, target_confidence: float = 0.8) -> tuple: 

144 

145 ''' 

146 Compute complete confidence filtering system with validation-based calibration. 

147 

148 Args: 

149 models (list): List of trained models for confidence estimation 

150 data (dict): Dictionary with validation and test data splits containing 'x_val', 'y_val', 'x_test', 'y_test', 'dt_test' keys 

151 target_confidence (float): Target percentage of predictions to classify as confident 

152 

153 Returns: 

154 tuple: Confidence threshold, filtered results, and calibration statistics 

155 ''' 

156 

157 # Step 1: Calibrate on validation data 

158 confidence_threshold, calibration_stats = calibrate_confidence_threshold( 

159 models, data['x_val'], data['y_val'], target_confidence 

160 ) 

161 

162 # Step 2: Apply to test data using calibrated threshold 

163 results = apply_confidence_filtering( 

164 models, data['x_test'], data['y_test'], confidence_threshold 

165 ) 

166 

167 # Step 3: Create detailed results DataFrame 

168 df_results = pl.DataFrame({ 

169 'datetime': data['dt_test'], 

170 'prediction': results['predictions'], 

171 'uncertainty': results['uncertainty'], 

172 'is_confident': results['confident_mask'], 

173 'confidence_threshold': confidence_threshold, 

174 'actual_value': data['y_test'], 

175 }) 

176 

177 # Add confidence scores (higher = more confident) 

178 df_results = df_results.with_columns([ 

179 (1 / (1 + pl.col('uncertainty'))).alias('confidence_score') 

180 ]) 

181 

182 # Sort by confidence score (most confident first) 

183 df_results = df_results.sort('confidence_score', descending=True) 

184 

185 return results, df_results, calibration_stats
